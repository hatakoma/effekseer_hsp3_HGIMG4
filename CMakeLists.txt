cmake_minimum_required(VERSION 3.20)
project(effekseer_hgimg4 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build with Visual Studio -A Win32 (HSP 3.7 is 32-bit)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(option BUILD_EXAMPLES BUILD_TOOLS BUILD_VIEWER BUILD_EDITOR BUILD_TEST BUILD_DX9 BUILD_DX11 BUILD_DX12 USE_OPENAL USE_DSOUND USE_XAUDIO2 NETWORK_ENABLED)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_GL ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/Effekseer EXCLUDE_FROM_ALL)
add_library(effekseer_hgimg4 SHARED src/bridge.cpp src/exports.def)
target_compile_definitions(effekseer_hgimg4 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(effekseer_hgimg4 PRIVATE EffekseerRendererGL opengl32)
target_compile_options(effekseer_hgimg4 PRIVATE /W4 /utf-8 /EHsc)
set_target_properties(effekseer_hgimg4 PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/dist")

# Build a separate ES2 renderer without changing the pinned upstream checkout.
# Upstream's Windows loader assumes WGL and requires ES3-only entry points.
get_target_property(gl_sources EffekseerRendererGL SOURCES)
get_target_property(gl_directory EffekseerRendererGL SOURCE_DIR)
set(angle_sources)
foreach(source IN LISTS gl_sources)
  get_filename_component(absolute "${source}" ABSOLUTE BASE_DIR "${gl_directory}")
  if(absolute MATCHES "EffekseerRendererGL.GLExtension.cpp$")
    file(READ "${absolute}" loader)
    string(REPLACE "wglGetProcAddress" "efk_angle_proc" loader "${loader}")
    foreach(entry glGenVertexArrays glDeleteVertexArrays glBindVertexArray glGenSamplers glDeleteSamplers glSamplerParameteri glBindSampler glMapBuffer glMapBufferRange glUnmapBuffer glDrawElementsInstanced glDrawBuffers glTexImage3D glCopyTexSubImage3D)
      string(REPLACE "GET_PROC_REQ(${entry})" "GET_PROC(${entry})" loader "${loader}")
    endforeach()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/angle_GLExtension.cpp" "${loader}")
    list(APPEND angle_sources "${CMAKE_CURRENT_BINARY_DIR}/angle_GLExtension.cpp")
  else()
    list(APPEND angle_sources "${absolute}")
  endif()
endforeach()
add_library(EffekseerRendererANGLE STATIC ${angle_sources})
target_include_directories(EffekseerRendererANGLE PUBLIC "${gl_directory}" PRIVATE "${gl_directory}/EffekseerRendererGL")
target_link_libraries(EffekseerRendererANGLE PUBLIC Effekseer EffekseerRendererCommon)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/exports.def" angle_exports)
string(REPLACE "LIBRARY effekseer_hgimg4" "LIBRARY effekseer_hgimg4dx" angle_exports "${angle_exports}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/angle_exports.def" "${angle_exports}")
add_library(effekseer_hgimg4dx SHARED src/bridge.cpp "${CMAKE_CURRENT_BINARY_DIR}/angle_exports.def" src/angle_loader.cpp src/angle_dispatch.cpp)
target_link_libraries(effekseer_hgimg4dx PRIVATE EffekseerRendererANGLE)
foreach(target EffekseerRendererANGLE effekseer_hgimg4dx)
  target_compile_definitions(${target} PRIVATE EFK_ANGLE __EFFEKSEER_RENDERER_GL2__ WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/src/angle_gl.h" /utf-8 /EHsc)
endforeach()
set_target_properties(effekseer_hgimg4dx PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/dist")
add_executable(collision_math_test tests/collision_math.cpp)
target_include_directories(collision_math_test PRIVATE src)
target_link_libraries(collision_math_test PRIVATE Effekseer)
target_compile_options(collision_math_test PRIVATE /utf-8 /EHsc)
enable_testing()
add_test(NAME collision_math COMMAND collision_math_test)
